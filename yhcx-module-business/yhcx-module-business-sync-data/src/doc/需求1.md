## 根据数据库表DDL创建Entity
```sql
CREATE TABLE `dataset_record_info` (
  `id` int NOT NULL AUTO_INCREMENT COMMENT '主键:数据集唯一ID',
  `tenant_id` bigint NOT NULL COMMENT '租户ID',
  `dataset_property` int DEFAULT '0' COMMENT '数据集类型（公有、私有）',
  `dataset_name` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '数据集/数据源名称',
  `dataset_storage_path` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '数据集/数据源ceph 存储路径',
  `dataset_type` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '数据集类型',
  `dataset_source` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT '' COMMENT '数据来源',
  `annotation_state` tinyint(1) DEFAULT '0' COMMENT '标注状态：0未标注，1已标注',
  `dataset_label` json DEFAULT NULL COMMENT '数据集标签列表',
  `dataset_size` bigint DEFAULT '0' COMMENT '数据源大小（BYTE）,视频是时长',
  `dataset_count` int DEFAULT '0' COMMENT '数据集数据量',
  `dataset_desc` varchar(2000) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '数据集描述',
  `gmt_create` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `gmt_modified` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '修改时间',
  `create_by` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '创建人',
  `modified_by` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '修改人',
  `deleted` tinyint(1) DEFAULT '0' COMMENT '是否删除，1删除，0未删除',
  `extend_info` json DEFAULT NULL COMMENT '扩展字段',
  `user_id` bigint NOT NULL COMMENT '用户ID',
  `dataset_scene` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT 'OTHER' COMMENT '数据场景',
  `upload_state` tinyint(1) DEFAULT '0' COMMENT '是否上传(0:未上传; 1:已上传;)',
  `frame_extraction` double(10,1) DEFAULT NULL COMMENT '抽帧频率，单位是：秒/帧',
  `status` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT 'NEW' COMMENT '数据集状态(启动中、上传中、上传成功、上传失败、取消上传、解析中、解析成功、解析失败)',
  `audio_split` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '音频切割规则',
  `parent_id` bigint DEFAULT NULL COMMENT '慧衍侧数据集id',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `idx_query_info` (`tenant_id`,`dataset_property`) USING BTREE,
  KEY `idx_date_info` (`gmt_modified`) USING BTREE
) ENGINE=InnoDB AUTO_INCREMENT=366 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci ROW_FORMAT=DYNAMIC COMMENT='据集配置管理记录';

```

CREATE TABLE `dataset_record_detail_info` (
  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键:导入元数据唯一标识',
  `tenant_id` bigint NOT NULL COMMENT '租户ID',
  `dataset_id` int NOT NULL COMMENT '数据源/数据集ID',
  `batch_id` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci NOT NULL COMMENT '批次ID',
  `file_name` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '上传文件名称',
  `file_type` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '文件扩展类型',
  `file_parent_path` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '文件父级目录',
  `need_compress` tinyint(1) DEFAULT '0' COMMENT '是否需要解压：0 否、1是',
  `storage_url` varchar(1024) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '上传文件存储地址',
  `error_msg` text CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci COMMENT '异常信息记录',
  `file_size` bigint DEFAULT '0' COMMENT '数据大小（Byte）',
  `file_status` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '文件上传状态 uploading 上传中，success上传成功 error上传失败',
  `upload_id` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '文件分片上传标识ID',
  `upload_progress` int DEFAULT '0' COMMENT '上传进度',
  `gmt_create` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `gmt_modified` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '修改时间',
  `create_by` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '创建人',
  `modified_by` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '修改人',
  `deleted` tinyint(1) DEFAULT '0' COMMENT '是否删除，1删除，0未删除',
  `user_id` bigint NOT NULL COMMENT '用户ID',
  `extend_info` json DEFAULT NULL COMMENT '扩展字段',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `idx_file_info` (`dataset_id`,`upload_id`) USING BTREE,
  KEY `idx_date_info` (`gmt_create`) USING BTREE
) ENGINE=InnoDB AUTO_INCREMENT=381 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci ROW_FORMAT=DYNAMIC COMMENT='数据导入明细';
